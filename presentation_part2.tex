\begin{frame}[plain, t]


\slideheader{Mục 01 • Tư Duy Thiết Kế}{Tư Duy Cốt Lõi Trước Khi Chọn Biểu Đồ}
\begin{columns}[T]
\begin{column}{0.54\textwidth}
{\fontsize{8.2pt}{10.5pt}\selectfont
Trước khi chọn loại biểu đồ, hãy trả lời 3 câu hỏi nền tảng:

\vspace{0.12cm}
\textbf{1. Quy luật hình ảnh có quan trọng với câu chuyện không?}
\begin{itemize}\setlength{\itemsep}{1.5pt}
    \item Nếu chỉ cần tra cứu vài con số cụ thể, một \textbf{bảng dữ liệu (Table)} hay một câu văn ngắn sẽ hiệu quả hơn.
    \item Chỉ vẽ biểu đồ khi cần làm nổi bật \textbf{xu hướng (trends)}, \textbf{hình thái phân bố (patterns)} hoặc \textbf{khoảng cách chênh lệch}.
\end{itemize}

\vspace{0.12cm}
\textbf{2. Hai yếu tố cốt lõi quyết định loại biểu đồ:}
\begin{itemize}\setlength{\itemsep}{1.5pt}
    \item \textbf{Định dạng dữ liệu (Data Format):} Chuỗi thời gian, danh mục, hay đa biến?
    \item \textbf{Câu chuyện muốn kể (Data Story):} So sánh nhóm, tỷ trọng, hay tương quan?
\end{itemize}
}
\end{column}
\begin{column}{0.44\textwidth}
\calloutbox[\textwidth]{Giằng Co: Tuân Thủ vs Phá Vỡ Quy Tắc (Lisa Charlotte Rost)}{
    • Các quy tắc giúp biểu đồ trung thực và dễ hiểu, nhưng áp dụng máy móc sẽ kìm hãm sáng tạo.\\[0.1cm]
    • Đôi khi các quy tắc tự mâu thuẫn: \textit{Làm sao vừa tạo đồ thị đơn giản dễ hiểu ngay, lại vừa phản ánh hết sự phức tạp của dữ liệu?}\\[0.1cm]
    $\rightarrow$ Hãy tự hỏi: \textbf{``Bạn muốn biểu đồ của mình được đánh giá dựa trên giá trị nào?''}
}
\end{column}

\begin{frame}[plain, t]

\begin{columns}[T]
\begin{column}{0.54\textwidth}
{\fontsize{7.8pt}{9.8pt}\selectfont
\textbf{1. Tiêu đề Kể chuyện vs. Kỹ thuật (Title \& Subtitle)}
\begin{itemize}\setlength{\itemsep}{0.8pt}
    \item \textit{Story Title:} Nêu bật thông điệp chính (\textit{``Đại dịch tấn công người Da đen nặng nhất''}).
    \item \textit{Technical Title:} Mô tả biến số (\textit{``Tử vong Covid-19 tại New York, 2020''}).
    \item $\rightarrow$ \textbf{Chiến lược kết hợp (Hybrid):} Story Title to đậm ở trên, Technical Title nhỏ ở dưới.
\end{itemize}

\vspace{0.05cm}
\textbf{2. Trục hoành ($x$), Trục tung ($y$), Chuỗi \& Điểm dữ liệu}
\begin{itemize}\setlength{\itemsep}{0.8pt}
    \item Xác định rõ thang đo, đơn vị và khoảng biến thiên.
\end{itemize}

\vspace{0.05cm}
\textbf{3. Nhãn trực tiếp (Labels) \& Chú thích (Annotations)}
\begin{itemize}\setlength{\itemsep}{0.8pt}
    \item Giải thích bối cảnh trực tiếp tại điểm ngoặt trên đồ thị.
\end{itemize}
}

\begin{itemize}\setlength{\itemsep}{2pt}
    \item Biểu đồ cột và thanh mã hóa giá trị thông qua \textbf{chiều dài} và \textbf{chiều cao} của hình chữ nhật.

\begin{itemize}\setlength{\itemsep}{2pt}
    \item Nếu trục bắt đầu từ $300$ thay vì $0$, cột $310$ sẽ trông như bẹp dí về $0$, phóng đại giả tạo chênh lệch (thủ thuật thao túng truyền thông).
\end{itemize}

\begin{frame}[plain, t]


\slideheader{Mục 02 • Nguyên Tắc Vàng}{Quy Tắc 2: Biểu Đồ Tròn (Pie Chart) Phải Đúng 100\%}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
{\fontsize{8.2pt}{10.5pt}\selectfont
Mắt người ước lượng diện tích góc quạt kém hơn chiều dài thanh. Nếu dùng Pie Chart, phải tuân thủ nghiêm ngặt:

\vspace{0.12cm}
\textbf{1. Tổng các phần bắt buộc phải bằng đúng 100\%}
\begin{itemize}\setlength{\itemsep}{1.5pt}
    \item Tổng khác 100\% là vô nghĩa.
    \item \textbf{Bẫy:} Khảo sát cho chọn nhiều đáp án hoặc chọn ``Tôi thích cả hai'' $\rightarrow$ tổng vượt 100\%, \textcolor{accentRed}{\textbf{tuyệt đối không vẽ biểu đồ tròn!}}
\end{itemize}

\vspace{0.12cm}
\textbf{2. Quy tắc thiết kế chuẩn xác:}
\begin{itemize}\setlength{\itemsep}{1.5pt}

\begin{itemize}\setlength{\itemsep}{1.5pt}
