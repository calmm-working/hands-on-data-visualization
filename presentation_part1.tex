    \textit{``Khi tỷ phú bước vào phòng, tính Mean ai cũng thành triệu phú, nhưng Median không ai giàu lên cả.''} $\rightarrow$ Dùng \textbf{Median} cho thu nhập, giá nhà.
}
}

\begin{frame}[plain, t]
\slideheader{Chương 5 • Ngôn Từ Chuẩn Xác}{\% Thay Đổi vs. Điểm Phần Trăm}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\calloutbox[\textwidth]{1. Thay Đổi Phần Trăm (Percent Change)}{
    Biến thiên tương đối so với giá trị ban đầu:\\[0.12cm]
    \centering
    $\mathbf{\Delta\% = \dfrac{V_{\text{mới}} - V_{\text{cũ}}}{|V_{\text{cũ}}|} \times 100\%}$
}

\vspace{0.12cm}
\calloutbox[\textwidth]{2. Chênh Lệch Điểm Phần Trăm (Percentage Point)}{
    Phép trừ số học giữa hai số tỷ lệ \%:\\[0.12cm]
    \centering
    $\mathbf{\text{Điểm \%} = \text{Tỷ lệ Mới} - \text{Tỷ lệ Cũ}}$
}
\end{column}
\begin{column}{0.48\textwidth}
\calloutbox[\textwidth]{Bẫy Truyền Thông: Tăng Thuế Tiêu Dùng (Sales Tax)}{
    Chính quyền tăng thuế từ \textbf{5\% lên 6\%} và tuyên bố: \textit{``Thuế chỉ tăng có 1\% thôi!''}\\[0.08cm]
    $\rightarrow$ \textbf{Tuyên bố này hoàn toàn SAI LỆCH!}\\[0.08cm]
    • \textbf{Nói đúng 1:} Tăng \textbf{1 điểm phần trăm} ($6\% - 5\% = 1\text{ pp}$).\\
    • \textbf{Nói đúng 2:} Gánh nặng thuế thực tế đã \textbf{tăng 20\%}:\\[0.10cm]
    \centering
    $\mathbf{\dfrac{6\% - 5\%}{5\%} = \dfrac{1\%}{5\%} = 20\%}$
}
\end{column}
\end{columns}

\end{frame}

\slideheader{Chương 5 • Ngôn Từ Chuẩn Xác}{Tương Quan Không Đồng Nghĩa Với Nhân Quả}
\begin{columns}[T]
\begin{column}{0.54\textwidth}
{\fontsize{8.2pt}{10.5pt}\selectfont
\textbf{Cực kỳ thận trọng với các động từ mang tính áp đặt:}
\textit{``dẫn đến'' (leads to)}, \textit{``thúc đẩy'' (promotes)}, \textit{``gây ra'' (causes)}.

\begin{itemize}\setlength{\itemsep}{2pt}
    \item \textbf{Tương quan (Correlation):} Hai biến số cùng biến thiên trên biểu đồ (ví dụ: tuổi thọ càng cao thì tỷ lệ sinh càng thấp).
    \item \textbf{Nhân quả (Causation):} Biến độc lập trực tiếp tạo ra sự thay đổi ở biến phụ thuộc.
\end{itemize}

\vspace{0.12cm}
\textbf{3 bước bắt buộc để khẳng định Quan hệ Nhân quả:}
\begin{enumerate}\setlength{\itemsep}{1.5pt}
    \item Chứng minh có \textbf{tương quan thống kê rõ rệt}.
    \item Xây dựng \textbf{lý thuyết cơ chế tác động} thuyết phục.
    \item \textbf{Cô lập hoàn toàn mọi biến gây nhiễu (Confounding variables)} -- yếu tố ẩn tác động đồng thời lên cả hai biến.
\end{enumerate}
}
\end{column}

Ngay cả khi không cố tình \textit{``chọn quả anh đào''} (cherry-picking), dữ liệu vẫn có thể bị bóp méo từ khâu lấy mẫu:

\vspace{0.08cm}
\textbf{1. Thiên vị chọn mẫu (Selection Bias)}
\begin{itemize}\setlength{\itemsep}{1pt}
    \item Mẫu được chọn khác biệt có hệ thống so với tổng thể.
    \item \textit{``Những gì bạn thấy phụ thuộc vào nơi bạn nhìn''} (Bergstrom \& West).
    \item \textbf{Ví dụ:} Đứng trước cửa phòng Gym để đo chiều cao ngẫu nhiên $\rightarrow$ kết luận sai rằng dân số cao vượt trội!
\end{itemize}

\vspace{0.08cm}
\textbf{2. Thiên vị không phản hồi (Non-response Bias)}
\begin{itemize}\setlength{\itemsep}{1pt}
    \item Người từ chối trả lời khác biệt người trả lời.
    \item \textbf{Thực tế:} Census 2020 trong đại dịch, người nghèo ít phản hồi $\rightarrow$ thu nhập trung vị bị đội lên cao giả tạo.
\end{itemize}
}

\begin{column}{0.52\textwidth}
{\fontsize{8.2pt}{10.5pt}\selectfont
\textbf{3. Thiên vị tự lựa chọn (Self-selection Bias)}
\begin{itemize}\setlength{\itemsep}{2pt}
    \item Xảy ra khi đánh giá hiệu quả chương trình nơi người tham gia \textbf{tự nguyện đăng ký} (giảm cân, học bổng).
    \item \textcolor{accentRed}{\textbf{So sánh SAI:}} So sánh người không tham gia (Nhóm A) với người tự đăng ký (Nhóm B).
    \item \textit{Lý do:} Nhóm B vốn đã có \textbf{động lực và ý thức cao hơn hẳn} Nhóm A!
\end{itemize}

\vspace{0.12cm}
\textbf{Giải pháp chuẩn (Randomized Trial):}
\begin{itemize}\setlength{\itemsep}{2pt}
    \item Lấy toàn bộ người đã tình nguyện đăng ký (Nhóm B -- có cùng mức quyết tâm).
    \item \textbf{Chia ngẫu nhiên (Randomly split)} thành Nhóm C (phác đồ 1) và Nhóm D (phác đồ 2) để so sánh công bằng.
\end{itemize}
