\begin{frame}[plain, t]

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\calloutbox[\linewidth]{1. Missing Values Nhập Sai}{
    Ô trống (\texttt{null/blank}) có ý nghĩa khác số \texttt{0}.\\[0.05cm]
    \textbf{Cảnh báo:} Người nhập điền \texttt{0}, \texttt{-1}, \texttt{-999} để báo khuyết làm sai lệch hoàn toàn hàm \texttt{SUM}, \texttt{AVERAGE}!
}
\vspace{0.15cm}
\calloutbox[\linewidth]{2. Mất Số 0 Dẫn Đầu (Leading Zeros)}{
    Xảy ra khi phần mềm tự ép kiểu Text sang Number.\\[0.05cm]
    \textbf{Hậu quả:} Mã Zip \texttt{06106} bị biến thành \texttt{6106}. Mã FIPS hạt LA \texttt{037} biến thành \texttt{37} (nhầm sang bang Bắc Carolina)!
}
\end{column}
\begin{column}{0.48\textwidth}
\calloutbox[\linewidth]{3. Chạm Giới Hạn Dòng Của Phần Mềm}{
    Nếu dữ liệu dừng ở dòng thứ \textbf{65.536} (giới hạn Excel \texttt{.xls} cũ) $\rightarrow$ Dữ liệu đã bị cắt cụt!\\[0.05cm]
    \textbf{Thực tế:} Y tế Anh từng mất hàng ngàn mẫu xét nghiệm Covid-19 vì lỗi này.
}
\vspace{0.15cm}
\calloutbox[\linewidth]{4. Bất Nhất Định Dạng Ngày Tháng}{
    Nhầm giữa chuẩn Mỹ \texttt{MM/DD/YYYY} và Quốc tế \texttt{DD/MM/YYYY}.\\[0.05cm]
    Xuất hiện ngày mặc định hệ thống (\texttt{1/1/1900}, \texttt{1970}) hoặc số thứ tự nội bộ Excel (\texttt{43891}).
}

\begin{frame}[plain, t]

\textbf{Cực kỳ thận trọng với các động từ mang tính áp đặt:}
\textit{``dẫn đến'' (leads to)}, \textit{``thúc đẩy'' (promotes)}, \textit{``gây ra'' (causes)}.

\begin{itemize}\setlength{\itemsep}{2pt}
    \item \textbf{Tương quan (Correlation):} Hai biến số cùng biến thiên trên biểu đồ (ví dụ: tuổi thọ càng cao thì tỷ lệ sinh càng thấp).
    \item \textbf{Nhân quả (Causation):} Biến độc lập trực tiếp tạo ra sự thay đổi ở biến phụ thuộc.
\end{itemize}

\vspace{0.12cm}
\textbf{3 bước bắt buộc để khẳng định Quan hệ Nhân quả:}
\begin{enumerate}\setlength{\itemsep}{1.5pt}
    \item Chứng minh có \textbf{tương quan thống kê rõ rệt}.
    \item Xây dựng \textbf{lý thuyết cơ chế tác động} thuyết phục.
    \item \textbf{Cô lập hoàn toàn mọi biến gây nhiễu (Confounding variables)} -- yếu tố ẩn tác động đồng thời lên cả hai biến.
\end{enumerate}
}

    \item Nếu chỉ cần tra cứu vài con số cụ thể, một \textbf{bảng dữ liệu (Table)} hay một câu văn ngắn sẽ hiệu quả hơn.
    \item Chỉ vẽ biểu đồ khi cần làm nổi bật \textbf{xu hướng (trends)}, \textbf{hình thái phân bố (patterns)} hoặc \textbf{khoảng cách chênh lệch}.
\end{itemize}

\vspace{0.12cm}
\textbf{2. Hai yếu tố cốt lõi quyết định loại biểu đồ:}
\begin{itemize}\setlength{\itemsep}{1.5pt}
    \item \textbf{Định dạng dữ liệu (Data Format):} Chuỗi thời gian, danh mục, hay đa biến?
    \item \textbf{Câu chuyện muốn kể (Data Story):} So sánh nhóm, tỷ trọng, hay tương quan?
\end{itemize}
}
\end{column}

    • Các quy tắc giúp biểu đồ trung thực và dễ hiểu, nhưng áp dụng máy móc sẽ kìm hãm sáng tạo.\\[0.1cm]

    $\rightarrow$ Hãy tự hỏi: \textbf{``Bạn muốn biểu đồ của mình được đánh giá dựa trên giá trị nào?''}

\vspace{0.12cm}
\textbf{1. Tuyến tính (Linear / Equal Interval):}
\begin{itemize}\setlength{\itemsep}{1.5pt}
    \item Chia đều khoảng cách min--max. Dành cho dữ liệu \textbf{phân bố chuẩn đối xứng}, làm nổi bật điểm cực đoan (outliers).

\begin{columns}[T]
\begin{column}{0.31\textwidth}
\calloutbox[\textwidth]{1. Thiên Vị Xác Nhận}{
    \textbf{Confirmation bias:}\\[0.05cm]
    • Não bộ chỉ tìm và tin dữ liệu khớp với định kiến có sẵn.\\[0.1cm]
    $\rightarrow$ \textbf{Khắc phục:} Chủ động tìm kiếm dữ liệu và cách diễn giải phản biện lại giả thuyết của mình.
}
\end{column}
\begin{column}{0.31\textwidth}
\calloutbox[\textwidth]{2. Thiên Vị Khuôn Mẫu}{
    \textbf{Pattern bias:}\\[0.05cm]
    • Não người luôn cố tìm quy luật ngay cả trong các con số ngẫu nhiên (nhiễu/noise).\\[0.1cm]
    $\rightarrow$ \textbf{Khắc phục:} Bắt buộc sử dụng kiểm định ý nghĩa thống kê trước khi kết luận.
}
\end{column}
\begin{column}{0.31\textwidth}
\calloutbox[\textwidth]{3. Thiên Vị Đóng Khung}{
    \textbf{Framing bias (Spiegelhalter):}\\[0.05cm]
    • \textit{``Tỷ lệ tử vong 5\%''} nghe đáng sợ hơn nhiều so với \textit{``Tỷ lệ sống sót 95\%''}.\\[0.08cm]
    • Nói thêm: \textit{``20 trên 400 người chết''} gây ám ảnh vì não hình dung ra con người cụ thể.
}
